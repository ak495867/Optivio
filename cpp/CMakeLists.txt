cmake_minimum_required(VERSION 3.20)
project(optivio_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(optivio_route_core route_core/route_core.cpp route_core/bridge.cpp)
target_compile_options(optivio_route_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
add_executable(test_route_core route_core/test_route_core.cpp)
target_link_libraries(test_route_core PRIVATE optivio_route_core)
add_executable(test_bridge route_core/test_bridge.cpp)
target_link_libraries(test_bridge PRIVATE optivio_route_core)
enable_testing()
add_test(NAME route_core COMMAND test_route_core)
add_test(NAME bridge COMMAND test_bridge)
